rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // 지금이 몇 번째 주인지 (월요일 0시 한국 시간 기준, 게임과 같은 계산)
    function weekNow() {
      return math.floor((math.floor((request.time.toMillis() + 32400000) / 86400000) + 3) / 7);
    }
    function mine(uid) {
      return request.auth != null && request.auth.uid == uid;
    }
    function okScore(d) {
      return d.keys().hasOnly(['w', 'n', 'st', 'net', 'sc', 'v', 'ts', 'live'])
        && d.keys().hasAll(['w', 'n', 'st', 'net', 'sc', 'v', 'ts'])
        && d.w == weekNow()
        && d.n is string && d.n.size() >= 1 && d.n.size() <= 10
        && d.st is int && d.st >= 1 && d.st <= 60
        && d.net is int && d.net >= 400 * d.st * d.st && d.net <= 800000 * d.st
        && d.sc == d.st * 1000000000 + d.net
        && d.v is int
        && d.ts == request.time
        && (!('live' in d) || d.live is bool);
    }

    // 주간 순위표: 누구나 읽기, 내 칸에만 쓰기, 더 좋은 기록일 때만 갱신, 지우기 불가
    match /weeks/{wid}/scores/{uid} {
      allow read: if true;
      allow create: if mine(uid) && okScore(request.resource.data);
      allow update: if mine(uid) && okScore(request.resource.data)
                    && request.resource.data.sc >= resource.data.sc;
      allow delete: if false;
    }

    // 다시 재생해서 확인할 수 있는 플레이 기록 (verify.html 에서 읽음)
    match /weeks/{wid}/proofs/{uid} {
      allow read: if true;
      allow create, update: if mine(uid)
        && request.resource.data.keys().hasOnly(['w', 'st', 'v', 'p'])
        && request.resource.data.w == weekNow()
        && request.resource.data.st is int
        && request.resource.data.v is int
        && request.resource.data.p is string
        && request.resource.data.p.size() < 400000;
      allow delete: if false;
    }

    // 가게 순위·구경: 누구나 읽기, 내 가게만 쓰기(좋아요 수는 못 바꿈), 좋아요는 한 사람 한 번
    function okShop(d) {
      return d.keys().hasOnly(['n', 'nk', 'g', 'sc', 'lk', 'room', 'eq', 'fur', 'ts'])
        && d.n is string && d.n.size() >= 1 && d.n.size() <= 12
        && d.nk is string && d.nk.size() <= 12
        && d.g is int && d.g >= 0 && d.g <= 6
        && d.sc is int && d.sc >= 0 && d.sc <= 3000000
        && d.lk is int && d.lk >= 0
        && d.room is int && d.room >= 4 && d.room <= 8
        && d.eq is map && d.eq.size() <= 16
        && d.fur is list && d.fur.size() <= 64
        && d.ts == request.time;
    }
    match /shops/{uid} {
      allow read: if true;
      allow create: if mine(uid) && okShop(request.resource.data) && request.resource.data.lk == 0;
      allow update: if (mine(uid) && okShop(request.resource.data) && request.resource.data.lk == resource.data.lk)
        || (request.auth != null && !mine(uid)
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['lk'])
            && request.resource.data.lk == resource.data.lk + 1
            && !exists(/databases/$(database)/documents/shops/$(uid)/likes/$(request.auth.uid))
            && existsAfter(/databases/$(database)/documents/shops/$(uid)/likes/$(request.auth.uid)));
      allow delete: if false;
      match /likes/{v} {
        allow read: if true;
        allow create: if request.auth != null && request.auth.uid == v && v != uid
          && getAfter(/databases/$(database)/documents/shops/$(uid)).data.lk == get(/databases/$(database)/documents/shops/$(uid)).data.lk + 1;
        allow update, delete: if false;
      }
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
