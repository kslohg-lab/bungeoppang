rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // 지금이 몇 번째 주인지 (월요일 0시 한국 시간 기준, 게임과 같은 계산)
    function weekNow() {
      return math.floor((math.floor((request.time.toMillis() + 32400000) / 86400000) + 3) / 7);
    }
    function mine(uid) {
      return request.auth != null && request.auth.uid == uid;
    }
    function okScore(d) {
      return d.keys().hasOnly(['w', 'n', 'st', 'net', 'sc', 'v', 'ts'])
        && d.keys().hasAll(['w', 'n', 'st', 'net', 'sc', 'v', 'ts'])
        && d.w == weekNow()
        && d.n is string && d.n.size() >= 1 && d.n.size() <= 10
        && d.st is int && d.st >= 1 && d.st <= 60
        && d.net is int && d.net >= 400 * d.st * d.st && d.net <= 800000 * d.st
        && d.sc == d.st * 1000000000 + d.net
        && d.v is int
        && d.ts == request.time;
    }

    // 주간 순위표: 누구나 읽기, 내 칸에만 쓰기, 더 좋은 기록일 때만 갱신, 지우기 불가
    match /weeks/{wid}/scores/{uid} {
      allow read: if true;
      allow create: if mine(uid) && okScore(request.resource.data);
      allow update: if mine(uid) && okScore(request.resource.data)
                    && request.resource.data.sc >= resource.data.sc;
      allow delete: if false;
    }

    // 다시 재생해서 확인할 수 있는 플레이 기록 (verify.html 에서 읽음)
    match /weeks/{wid}/proofs/{uid} {
      allow read: if true;
      allow create, update: if mine(uid)
        && request.resource.data.keys().hasOnly(['w', 'st', 'v', 'p'])
        && request.resource.data.w == weekNow()
        && request.resource.data.st is int
        && request.resource.data.v is int
        && request.resource.data.p is string
        && request.resource.data.p.size() < 400000;
      allow delete: if false;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
